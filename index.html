    function atualizarDashboard() {

      let total = 0, conc = 0, min = 0, ac = 0, totQuest = 0;

      meusDados.forEach(d => {

        d.topicos.forEach(t => {

            total++;

            min += t.tempo;

            ac += t.acertos;

            totQuest += t.totais;

        });

      });

      document.getElementById('stat-progresso').innerText = total > 0 ? Math.round((conc/total)*100)+'%' : '0%';

      document.getElementById('stat-topicos').innerText = `${conc} de ${total} tópicos`;

      document.getElementById('stat-desempenho').innerText = totQuest > 0 ? Math.round((ac/totQuest)*100)+'%' : '0%';

      document.getElementById('stat-questoes').innerText = `${ac} acertos / ${totQuest} feitas`;

      document.getElementById('stat-tempo').innerText = `${Math.floor(min/60)}h ${min%60}m`;

    }

    function calcularDiagnostico() {

        let total = 0, conc = 0;

        meusDados.forEach(d => d.topicos.forEach(t => { total++; if (t.concluido) conc++; }));

        let faltam = total - conc;

       

        let alvo = new Date(minhaEstrategia.dataAlvo);

        let hoje = new Date();

        let diasRestantes = Math.ceil((alvo - hoje) / (1000 * 60 * 60 * 24));

        if (diasRestantes < 1) diasRestantes = 1;

        let ritmoNecessario = Math.ceil(faltam / diasRestantes);

        let ritmoAtual = minhaEstrategia.topicos;

        document.getElementById('diag-pendentes').innerText = faltam;

        document.getElementById('diag-ritmo').innerHTML = `${ritmoNecessario} <span class="text-sm font-normal text-gray-500">/dia</span>`;

        let msg = "", cor = "";

        if (conc >= total && total > 0) {

            msg = "<i class='fa-solid fa-check-circle mr-2'></i> <strong>Edital 100% finalizado!</strong> Foco total em revisões e simulados. Você está pronto.";

            cor = "text-emerald-400";

        } else if (ritmoAtual >= ritmoNecessario) {

            msg = `<i class='fa-solid fa-rocket mr-2'></i> <strong>Estratégia Impecável!</strong> Seu ritmo planejado de ${ritmoAtual} tópicos/dia é superior à exigência matemática (${ritmoNecessario}/dia). Mantendo a disciplina, você finalizará o edital ANTES da data alvo.`;

Sua meta configurada é de apenas ${ritmoAtual}/dia. Reajuste seus parâmetros ou estenda a data.`;

            cor = "text-yellow-400";

        }

        document.getElementById('diag-alerta').innerHTML = `<span class="${cor}">${msg}</span>`;

        document.getElementById('missao-lista').innerHTML = `

            <label class="flex items-center gap-3 p-3 bg-[#1a1d24] rounded-lg border border-gray-700 cursor-pointer hover:border-brand transition">

    function renderizarMuralRevisoes() {

        const cAtrasada = document.getElementById('rev-atrasadas');

        const cHoje = document.getElementById('rev-hoje');

        const cFuturas = document.getElementById('rev-futuras');

       

        cAtrasada.innerHTML = cHoje.innerHTML = cFuturas.innerHTML = '';

        const hojeObj = new Date(); hojeObj.setHours(0,0,0,0);

        minhasRevisoes.forEach((rev, idx) => {

            if(rev.status === 'concluida') return;

            const dataRev = new Date(rev.dataAgendada + 'T00:00:00');

            const diffDays = Math.ceil((dataRev - hojeObj) / (1000 * 60 * 60 * 24));

            const disc = meusDados.find(d => d.id === rev.discId);

            const topico = disc ? disc.topicos[rev.topicoIdx] : null;

            if(!topico) return;

            let cor = 'blue', destino = cFuturas, txtData = `Em ${diffDays} dias`;

            if(diffDays < 0) { cor = 'red'; destino = cAtrasada; txtData = `Atrasada ${Math.abs(diffDays)} dias`; }

            else if(diffDays === 0) { cor = 'emerald'; destino = cHoje; txtData = 'Revisar Hoje'; }

            const card = document.createElement('div');

            card.className = `p-4 rounded-lg border border-gray-700 bg-[#1a1d24] border-l-4 border-l-${cor}-500 shadow-sm hover:border-gray-500 transition`;

            card.innerHTML = `

                <p class="text-xs text-${cor}-400 font-bold mb-1">${txtData}</p>

                <p class="text-xs text-gray-400 font-semibold uppercase">${disc.nome}</p>

                <p class="text-sm font-medium text-gray-200 mt-1 mb-3 line-clamp-2" title="${topico.nome}">${topico.nome}</p>

                <div class="flex gap-1.5">

                    <button onclick="concluirRevisao(${idx}, 7)" class="flex-1 bg-gray-800 hover:bg-gray-700 text-xs py-1.5 rounded transition font-medium" title="Concluir e agendar próxima para 7 dias">+7d</button>

                    <button onclick="concluirRevisao(${idx}, 15)" class="flex-1 bg-gray-800 hover:bg-gray-700 text-xs py-1.5 rounded transition font-medium" title="Concluir e agendar próxima para 15 dias">+15d</button>

                    <button onclick="concluirRevisao(${idx}, 30)" class="flex-1 bg-gray-800 hover:bg-gray-700 text-xs py-1.5 rounded transition font-medium" title="Concluir e agendar próxima para 30 dias">+30d</button>

                    <button onclick="apagarRevisao(${idx})" class="w-8 flex-shrink-0 bg-red-900/30 hover:bg-red-900/50 text-red-400 text-xs py-1.5 rounded transition" title="Excluir revisão"><i class="fa-solid fa-trash"></i></button>

                </div>

            `;

            destino.appendChild(card);

        });

        if(cAtrasada.innerHTML==='') cAtrasada.innerHTML = '<p class="text-xs text-gray-600 italic">Nenhuma revisão atrasada!</p>';

        if(cHoje.innerHTML==='') cHoje.innerHTML = '<p class="text-xs text-gray-600 italic">Tudo em dia para hoje!</p>';

        if(cFuturas.innerHTML==='') cFuturas.innerHTML = '<p class="text-xs text-gray-600 italic">Nenhuma revisão programada.</p>';

    }

    function concluirRevisao(idx, proximosDias) {

        minhasRevisoes[idx].status = 'concluida';

        const dataNova = new Date(); dataNova.setDate(dataNova.getDate() + proximosDias);

        minhasRevisoes.push({ id: Date.now(), discId: minhasRevisoes[idx].discId, topicoIdx: minhasRevisoes[idx].topicoIdx, dataAgendada: dataNova.toISOString().split('T')[0], status: 'pendente' });

        salvarTudo();

    }

    function apagarRevisao(idx) { minhasRevisoes[idx].status = 'concluida'; salvarTudo(); }

    function salvarModal() {

        const dId = document.getElementById('modal-disc').value;

        const tIdx = parseInt(document.getElementById('modal-topico').value);

        const discIndex = meusDados.findIndex(d => d.id === dId);

       

        if(modalMode === 'estudo') {

            const min = parseInt(document.getElementById('modal-tempo').value) || 0;

            const ac = parseInt(document.getElementById('modal-acertos').value) || 0;

            const tot = parseInt(document.getElementById('modal-totais').value) || 0;

           

            // Atualiza Streak se estudou >= 20 minutos

            if (min >= 20) {

                let streakAtual = parseInt(localStorage.getItem('streak_atual_amanda') || 0);

                let recorde = parseInt(localStorage.getItem('streak_recorde_amanda') || 0);

                streakAtual += 1;

                if(streakAtual > recorde) recorde = streakAtual;

                localStorage.setItem('streak_atual_amanda', streakAtual);

                localStorage.setItem('streak_recorde_amanda', recorde);

                renderizarStreak();

            }

            meusDados[discIndex].topicos[tIdx].tempo += min;

            meusDados[discIndex].topicos[tIdx].acertos += ac;

            meusDados[discIndex].topicos[tIdx].totais += tot;

        } else {

            const dias = parseInt(document.getElementById('modal-dias-rev').value) || 7;

            const data = new Date(); data.setDate(data.getDate() + dias);

            minhasRevisoes.push({ id: Date.now(), discId: dId, topicoIdx: tIdx, dataAgendada: data.toISOString().split('T')[0], status: 'pendente' });

        }

       

        salvarTudo();

        closeModalEstudo();

        document.getElementById('modal-tempo').value = '';

        document.getElementById('modal-acertos').value = '';

        document.getElementById('modal-totais').value = '';

    }
